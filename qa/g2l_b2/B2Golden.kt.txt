package com.thelastjailer.app.checkpoint

import com.thelastjailer.app.*

private fun dump(tag: String, rig: Rig, out: StringBuilder) {
    out.append("## ").append(tag).append('\n')
    for (name in listOf(Story.LIVE) + (1..3).map { Cp.file(it) }) {
        val m = rig.device.disk(name)
        out.append("[").append(name).append("]\n")
        m.entries.sortedBy { it.key }.forEach { (k, v) ->
            val r = if (v is Set<*>) v.map { it.toString() }.sorted().toString() else v.toString()
            // checkpoint files carry random lineage ids; normalise them so runs compare
            out.append(k).append('=').append(v?.javaClass?.simpleName).append(':').append(r).append('\n')
        }
    }
}

fun main() {
    val out = StringBuilder()
    for (withRecorder in listOf(true, false)) {
        val sfx = if (withRecorder) "+rec" else "-rec"
        fun rig() = Rig(withRecorder = withRecorder, hooks = ArrivalRecorder.Hooks(newLineage = { String.format("%016x", (counter++).toLong() + 0xabc000L) }))
        counter = 0
        var r = rig()
        r.playTo(6); dump("playTo6$sfx", r, out)
        counter = 0; r = rig()
        val (n, c) = Story.staying(2)
        val st = Story.stateAt(n, 1, "s")
        r.saves.commitChoice(st, c, r.recorder); dump("staying$sfx", r, out)
        r.saves.restorePreviousState(1); dump("restore$sfx", r, out)
        counter = 0; r = rig()
        val enc = CombatEncounter(id = "e", enemyId = "x", victoryNodeId = "after_victory", defeatNodeId = "after_defeat", xpReward = 10, goldReward = 5, unlockTrophy = "T")
        val s0 = Story.stateAt(n, 1, "c")
        r.saves.commitChoice(s0, c, r.recorder)
        r.saves.commitCombatResult(s0.copy(sceneId = c.nextNodeId), enc, CombatOutcome(true, 3, emptyList()), r.recorder); dump("combat$sfx", r, out)
        r.saves.commitPurchase(r.saves.load(1)!!.copy(gold = 90), "healing_draught", 5, r.recorder); dump("purchase$sfx", r, out)
        r.saves.commitSlotSave(1, r.saves.load(1)!!); dump("saveSame$sfx", r, out)
        r.saves.commitSlotSave(2, r.saves.load(1)!!); dump("saveOther$sfx", r, out)
        r.saves.setActiveSlot(1); dump("activeSlot$sfx", r, out)
        r.saves.commitRestart(2, r.recorder); dump("restart$sfx", r, out)
        counter = 0; r = rig()
        r.playTo(4); r.saves.commitRestart(1, r.recorder); dump("playRestart$sfx", r, out)
        counter = 0; r = rig()
        for ((i, loop) in Story.loops().withIndex()) { val b = Story.stateAt(loop.first, 1, "l$i"); r.saves.commitChoice(b, loop.second, r.recorder) }
        dump("loops$sfx", r, out)
        counter = 0; r = Rig(withRecorder = withRecorder, enabled = false)
        r.playTo(3); dump("disabled$sfx", r, out)
    }

    run {
        counter = 0
        val device = SimulatedDevice()
        val r0 = Rig(device, hooks = ArrivalRecorder.Hooks(newLineage = { String.format("%016x", (counter++).toLong() + 0xabc000L) }))
        device.armFault(Cp.file(1), 1, CommitFault.FALSE_UNCHANGED)
        r0.arrive(4)
        device.writeDisk(Cp.file(1), mapOf("h.version" to CHECKPOINT_FORMAT_VERSION + 1, "h.slot" to 1, "h.lineage" to Cp.L1, "c05.future" to "data", "c05.crc" to 7L))
        val re = r0.reopen()
        val live = re.saves.load(1)!!
        re.saves.commitChoice(live, Story.staying(5).second, re.recorder)
        dump("flipOff", re, out)
        re.saves.commitChoice(re.saves.load(1)!!, Story.crossing(5).second, re.recorder)
        dump("flipOffThenArrive", re, out)
    }

    run {
        counter = 0
        val device = SimulatedDevice()
        val r0 = Rig(device, hooks = ArrivalRecorder.Hooks(newLineage = { String.format("%016x", (counter++).toLong() + 0xabc000L) }))
        device.armFault(Cp.file(1), 1, CommitFault.FALSE_UNCHANGED)
        r0.arrive(4)
        device.writeDisk(Cp.file(1), mapOf("h.version" to CHECKPOINT_FORMAT_VERSION + 1, "h.slot" to 1, "h.lineage" to Cp.L1, "c05.future" to "data", "c05.crc" to 7L))
        val (node, choice) = Story.crossing(5)
        r0.saves.commitChoice(Story.stateAt(node, 1, "a"), choice, r0.recorder)
        dump("flipDuringArrival", r0, out)
    }
    println(out)
}
private var counter = 0
